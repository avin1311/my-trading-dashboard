    29→// ==================== POWER BI PANEL WRAPPER ====================
    30→function Panel({ title, icon: Icon, badge, children, className, source, span }: {
    31→  title: string; icon?: React.ElementType; badge?: React.ReactNode;
    32→  children: React.ReactNode; className?: string; source?: string;
    33→  span?: string;
    34→}) {
    35→  return (
    36→    <div className={cn(
    37→      'rounded-xl border border-slate-800/60 bg-[#0d1017]/80 backdrop-blur-sm overflow-hidden flex flex-col',
    38→      span && span,
    39→      className
    40→    )}>
    41→      <div className="flex items-center justify-between px-4 py-2.5 border-b border-slate-800/40 bg-slate-900/30 shrink-0">
    42→        <div className="flex items-center gap-2">
    43→          {Icon && <Icon className="w-3.5 h-3.5 text-slate-500" />}
    44→          <span className="text-[11px] font-semibold text-slate-300 uppercase tracking-wider">{title}</span>
    45→          {source && (
    46→            <Badge variant="outline" className="text-[7px] px-1 py-0 bg-blue-500/8 border-blue-500/20 text-blue-400/70 h-3.5">{source}</Badge>
    47→          )}
    48→        </div>
    49→        {badge}
    50→      </div>
    51→      <div className="p-3.5 flex-1 min-h-0">{children}</div>
    52→    </div>
    53→  );
    54→}

    59→    <div className="text-center p-2 rounded-lg bg-slate-800/20 border border-slate-800/30">

    64→  );
