    58→    58→function fINR(v: number): string {
    59→    59→  if (v >= 1e12) return '\u20B9' + (v / 1e12).toFixed(2) + ' T';
    60→    60→  if (v >= 1e7) return '\u20B9' + (v / 1e7).toFixed(2) + ' Cr';
    61→    61→  if (v >= 1e5) return '\u20B9' + (v / 1e5).toFixed(2) + ' L';
    62→    62→  return '\u20B9' + v.toLocaleString('en-IN', { minimumFractionDigits: 2, maximumFractionDigits: 2 });
    63→    63→}
    64→    64→function fNum(v: number): string {
    65→    65→  if (v >= 1e12) return (v / 1e12).toFixed(2) + 'T';
    66→    66→  if (v >= 1e7) return (v / 1e7).toFixed(2) + ' Cr';
    67→    67→  if (v >= 1e5) return (v / 1e5).toFixed(2) + ' L';
    68→    68→  if (v >= 1e3) return (v / 1e3).toFixed(1) + 'K';
    69→    69→  return v.toFixed(2);
    70→    70→}
    71→    71→function fDate(d: string): string {
    72→    72→  return new Date(d).toLocaleDateString('en-IN', { day: '2-digit', month: 'short', year: '2-digit' });
    73→    73→}
    74→    74→function fTime(d: string): string {
    75→    75→  const dt = new Date(d);
    76→    76→  const now = new Date();
    77→    77→  const diffMs = now.getTime() - dt.getTime();
    78→    78→  const diffHrs = Math.floor(diffMs / 3600000);
    79→    79→  if (diffHrs < 1) return 'Just now';
    80→    80→  if (diffHrs < 24) return diffHrs + 'h ago';
    81→    81→  return fDate(d);
    82→    82→}

   210→   210→function OwnershipDonut({ data }: { data: Record<string, number | null> }) {
   211→   211→  const segments = [
   212→   212→    { label: 'Promoter', value: data.promoter, color: '#3b82f6' },
   213→   213→    { label: 'FII', value: data.fii, color: '#10b981' },
   214→   214→    { label: 'DII', value: data.dii, color: '#f59e0b' },
   215→   215→    { label: 'Public', value: data.public, color: '#6366f1' },
   216→   216→  ].filter(s => s.value !== null && s.value !== undefined) as { label: string; value: number; color: string }[];
   217→   217→  const total = segments.reduce((a, s) => a + s.value, 0);
   218→   218→  if (total === 0) return <div className="text-xs text-slate-500 text-center py-4">No ownership data</div>;
   219→   219→
   220→   220→  let cumulativePercent = 0;
   221→   221→  const gradientParts = segments.map(s => {
   222→   222→    const start = cumulativePercent;
   223→   223→    cumulativePercent += (s.value / total) * 100;
   224→   224→    return `${s.color} ${start}% ${cumulativePercent}%`;
   225→   225→  });
   226→   226→
   227→   227→  return (
   228→   228→    <div className="flex items-center gap-4">
   229→   229→      <div className="relative w-24 h-24 shrink-0">
   230→   230→        <div className="w-24 h-24 rounded-full" style={{ background: `conic-gradient(${gradientParts.join(', ')})` }} />
   231→   231→        <div className="absolute inset-3 rounded-full bg-slate-900 flex items-center justify-center">
   232→   232→          <span className="text-[10px] text-slate-400 font-semibold">Holding</span>
   233→   233→        </div>
   234→   234→      </div>
   235→   235→      <div className="space-y-1.5 flex-1">
   236→   236→        {segments.map(s => (
   237→   237→          <div key={s.label} className="flex items-center gap-2 text-xs">
   238→   238→            <div className="w-2.5 h-2.5 rounded-sm shrink-0" style={{ background: s.color }} />
   239→   239→            <span className="text-slate-400 flex-1">{s.label}</span>
   240→   240→            <span className="font-mono text-slate-200 font-semibold">{s.value.toFixed(1)}%</span>
   241→   241→          </div>
   242→   242→        ))}
   243→   243→      </div>
   244→   244→    </div>
   245→   245→  );
   246→   246→}
