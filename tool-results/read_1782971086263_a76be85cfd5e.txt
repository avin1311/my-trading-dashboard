    58→// ==================== FORMATTERS ====================
    59→function fINR(v: number): string {
    60→  if (v >= 1e12) return '\u20B9' + (v / 1e12).toFixed(2) + ' T';
    61→  if (v >= 1e7) return '\u20B9' + (v / 1e7).toFixed(2) + ' Cr';
    62→  if (v >= 1e5) return '\u20B9' + (v / 1e5).toFixed(2) + ' L';
    63→  return '\u20B9' + v.toLocaleString('en-IN', { minimumFractionDigits: 2, maximumFractionDigits: 2 });
    64→}
    65→function fNum(v: number): string {
    66→  if (v >= 1e12) return (v / 1e12).toFixed(2) + 'T';
    67→  if (v >= 1e7) return (v / 1e7).toFixed(2) + ' Cr';
    68→  if (v >= 1e5) return (v / 1e5).toFixed(2) + ' L';
    69→  if (v >= 1e3) return (v / 1e3).toFixed(1) + 'K';
    70→  return v.toFixed(2);
    71→}
    72→function fDate(d: string): string { return new Date(d).toLocaleDateString('en-IN', { day: '2-digit', month: 'short', year: '2-digit' }); }
    73→function fTime(d: string): string { const dt = new Date(d); const now = new Date(); const diffHrs = Math.floor((now.getTime() - dt.getTime()) / 3600000); if (diffHrs < 1) return 'Just now'; if (diffHrs < 24) return diffHrs + 'h ago'; return fDate(d); }

   141→function PBar({ value, min, max, label, color }: { value: number; min: number; max: number; label: string; color: string }) {

   144→}

   151→function OwnershipDonut({ data }: { data: Record<string, number | null> }) {
   152→  const segments = [{ label: 'Promoter', value: data.promoter, color: '#3b82f6' }, { label: 'FII', value: data.fii, color: '#10b981' }, { label: 'DII', value: data.dii, color: '#f59e0b' }, { label: 'Public', value: data.public, color: '#6366f1' }].filter(s => s.value !== null && s.value !== undefined) as { label: string; value: number; color: string }[];
   153→  const total = segments.reduce((a, s) => a + s.value, 0);
   154→  if (total === 0) return <div className="text-xs text-slate-500 text-center py-4">No ownership data</div>;
   155→  let cum = 0;
   156→  const grad = segments.map(s => { const start = cum; cum += (s.value / total) * 100; return `${s.color} ${start}% ${cum}%`; });
   157→  return (<div className="flex items-center gap-4"><div className="relative w-24 h-24 shrink-0"><div className="w-24 h-24 rounded-full" style={{ background: `conic-gradient(${grad.join(', ')})` }} /><div className="absolute inset-3 rounded-full bg-slate-900 flex items-center justify-center"><span className="text-[10px] text-slate-400 font-semibold">Holding</span></div></div><div className="space-y-1.5 flex-1">{segments.map(s => (<div key={s.label} className="flex items-center gap-2 text-xs"><div className="w-2.5 h-2.5 rounded-sm shrink-0" style={{ background: s.color }} /><span className="text-slate-400 flex-1">{s.label}</span><span className="font-mono text-slate-200 font-semibold">{s.value.toFixed(1)}%</span></div>))}</div></div>);
   158→}
