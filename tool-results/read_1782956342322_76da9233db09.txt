   200→function SentimentBadge({ sentiment }: { sentiment: string }) {
   201→  const map: Record<string, string> = {
   202→    positive: 'bg-emerald-500/15 text-emerald-400 border-emerald-500/30',
   203→    negative: 'bg-red-500/15 text-red-400 border-red-500/30',
   204→    neutral: 'bg-slate-500/15 text-slate-400 border-slate-500/30',
   205→  };
   206→  return <Badge variant="outline" className={cn('text-[8px] px-1.5 py-0 capitalize', map[sentiment] || map.neutral)}>{sentiment}</Badge>;
   207→}

   209→// ==================== OWNERSHIP PIE (CSS-only) ====================
   210→function OwnershipDonut({ data }: { data: Record<string, number | null> }) {
   211→  const segments = [
   212→    { label: 'Promoter', value: data.promoter, color: '#3b82f6' },
   213→    { label: 'FII', value: data.fii, color: '#10b981' },
   214→    { label: 'DII', value: data.dii, color: '#f59e0b' },
   215→    { label: 'Public', value: data.public, color: '#6366f1' },
   216→  ].filter(s => s.value !== null && s.value !== undefined) as { label: string; value: number; color: string }[];
   217→  const total = segments.reduce((a, s) => a + s.value, 0);
   218→  if (total === 0) return <div className="text-xs text-slate-500 text-center py-4">No ownership data</div>;
   219→
   220→  let cumulativePercent = 0;
   221→  const gradientParts = segments.map(s => {
   222→    const start = cumulativePercent;
   223→    cumulativePercent += (s.value / total) * 100;
   224→    return `${s.color} ${start}% ${cumulativePercent}%`;
   225→  });
   226→
   227→  return (
   228→    <div className="flex items-center gap-4">
   229→      <div className="relative w-24 h-24 shrink-0">
   230→        <div className="w-24 h-24 rounded-full" style={{ background: `conic-gradient(${gradientParts.join(', ')})` }} />
   231→        <div className="absolute inset-3 rounded-full bg-slate-900 flex items-center justify-center">
   232→          <span className="text-[10px] text-slate-400 font-semibold">Holding</span>
   233→        </div>
   234→      </div>
   235→      <div className="space-y-1.5 flex-1">
   236→        {segments.map(s => (
   237→          <div key={s.label} className="flex items-center gap-2 text-xs">
   238→            <div className="w-2.5 h-2.5 rounded-sm shrink-0" style={{ background: s.color }} />
   239→            <span className="text-slate-400 flex-1">{s.label}</span>
   240→            <span className="font-mono text-slate-200 font-semibold">{s.value.toFixed(1)}%</span>
   241→          </div>
   242→        ))}
   243→      </div>
   244→    </div>
   245→  );
   246→}
