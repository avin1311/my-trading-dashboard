   603→      return 0;
   604→    });
   605→    return arr;

   983→// ==================== WATCHLIST VIEW ====================
   984→function WatchlistView({ d, watchlist }: { d: ReturnType<typeof useDashboardData>; watchlist: ReturnType<typeof useWatchlist> }) {
   985→  const [watchlistQuotes, setWatchlistQuotes] = useState<Record<string, { price: number; changePct: number; name: string; loading: boolean }>>({});
   986→  useEffect(() => {
   987→    if (watchlist.watchlist.length === 0) return;
   988→    const fetchQuotes = async () => {
   989→      const results: typeof watchlistQuotes = {};
   990→      for (const item of watchlist.watchlist) {
   991→        results[item.symbol] = { price: 0, changePct: 0, name: item.symbol, loading: true };
   992→      }
   993→      setWatchlistQuotes({ ...results });
   994→      for (const item of watchlist.watchlist) {
   995→        try {
   996→          const res = await fetch('/api/stock-detail?symbol=' + item.symbol);
   997→          const data = await res.json();
   998→          if (data.quote) {
   999→            results[item.symbol] = { price: data.quote.price, changePct: data.quote.changePct, name: data.quote.longName || data.quote.name, loading: false };
  1000→          }
  1001→        } catch { results[item.symbol] = { ...results[item.symbol], loading: false }; }
  1002→        setWatchlistQuotes({ ...results });
  1003→      }
  1004→    };
  1005→    fetchQuotes();
  1006→    const interval = setInterval(fetchQuotes, 30000);
  1007→    return () => clearInterval(interval);
  1008→  }, [watchlist.watchlist]);
  1009→  return (
  1010→    <P title="My Watchlist" icon={Star} badge={<Badge variant="outline" className="text-[10px] px-2 py-0.5 bg-amber-500/10 border-amber-500/30 text-amber-400">{watchlist.watchlist.length} stocks</Badge>} source="Custom">
  1011→      {watchlist.watchlist.length === 0 ? (
  1012→        <div className="text-center py-16">
  1013→          <Star className="w-12 h-12 mx-auto mb-3 text-slate-700" />
  1014→          <p className="text-sm text-slate-400">Your watchlist is empty</p>
  1015→          <p className="text-xs text-slate-600 mt-1">Click the star icon on any stock to add it here</p>
  1016→        </div>
  1017→      ) : (
  1018→        <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 xl:grid-cols-4 gap-3">
  1019→          {watchlist.watchlist.map(item => {
  1020→            const info = watchlistQuotes[item.symbol];
  1021→            return (
  1022→              <div key={item.symbol} className="p-3 rounded-xl bg-slate-800/20 border border-slate-800/40 hover:bg-slate-800/40 cursor-pointer transition-colors" onClick={() => d.handleSelect(item.symbol, item.type)}>
  1023→                <div className="flex items-center justify-between mb-2">
  1024→                  <div>
  1025→                    <div className="text-sm font-bold text-slate-200">{item.symbol}</div>
  1026→                    <div className="text-[10px] text-slate-500 truncate max-w-[140px]">{info?.name || item.symbol}</div>
  1027→                  </div>
  1028→                  <button onClick={(e) => { e.stopPropagation(); watchlist.removeFromWatchlist(item.symbol); }} className="p-1 rounded hover:bg-red-500/20 text-slate-600 hover:text-red-400">
  1029→                    <X className="w-3.5 h-3.5" />
  1030→                  </button>
  1031→                </div>
  1032→                {info?.loading ? <Skeleton className="h-6 w-24 bg-slate-700" /> : info?.price ? (
  1033→                  <div className="flex items-end justify-between">
  1034→                    <span className="text-lg font-bold font-mono text-slate-100">{info.price.toLocaleString('en-IN', { maximumFractionDigits: 2 })}</span>
  1035→                    <span className={cn('text-sm font-semibold font-mono', info.changePct >= 0 ? 'text-emerald-400' : 'text-red-400')}>
  1036→                      {info.changePct >= 0 ? '+' : ''}{info.changePct.toFixed(2)}%
  1037→                    </span>
  1038→                  </div>
  1039→                ) : <Skeleton className="h-6 w-24 bg-slate-700" />}
  1040→              </div>
  1041→            );
  1042→          })}
  1043→        </div>
  1044→      )}
  1045→    </P>
  1046→  );
  1047→}
