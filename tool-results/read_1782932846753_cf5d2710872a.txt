   257→// ==================== HTTP HELPER ====================
   258→function httpsGet(url: string): Promise<string> {
   259→  return new Promise((resolve, reject) => {
   260→    const https = require('https');
   261→    const req = https.get(url, {
   262→      headers: { 'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/120.0.0.0 Safari/537.36' }
   263→    }, (res: any) => {
   264→      if (res.statusCode === 301 || res.statusCode === 302) {
   265→        httpsGet(res.headers.location).then(resolve).catch(reject);
   266→        return;
   267→      }
   268→      let data = '';
   269→      res.on('data', (c: string) => (data += c));
   270→      res.on('end', () => resolve(data));
   271→    });
   272→    req.on('error', reject);
   273→    req.setTimeout(10000, () => { req.destroy(); reject(new Error('Timeout')); });
   274→  });
   275→}

   290→// ==================== CALCULATE MOVING AVERAGES ====================
   291→function calculateSMA(data: number[], period: number): number | null {
   292→  if (data.length < period) return null;
   293→  const slice = data.slice(-period);
   294→  return slice.reduce((a, b) => a + b, 0) / period;

   401→// ==================== HISTORICAL DATA ====================
   402→export async function getHistoricalData(
   403→  nseSymbol: string,
   404→  days: number = 200
   405→): Promise<HistoricalDataPoint[]> {
   406→  const cacheKey = `${nseSymbol}_${days}`;
   407→  const cached = histCache.get(cacheKey);
   408→  if (cached && Date.now() - cached.timestamp < HIST_TTL) {
   409→    return cached.data;
   410→  }
   411→
   412→  const yahooSymbol = getYahooSymbol(nseSymbol);
   413→  const period1 = Math.floor(new Date(Date.now() - Math.ceil(days * 1.5) * 86400000).getTime() / 1000);
   414→  const url = `https://query1.finance.yahoo.com/v8/finance/chart/${encodeURIComponent(yahooSymbol)}?period1=${period1}&period2=9999999999&interval=1d`;
   415→
   416→  const body = await httpsGet(url);
   417→  const data = JSON.parse(body);
   418→  const result = data.chart?.result?.[0];
   419→
   420→  if (!result) throw new Error(`No historical data for ${nseSymbol}`);
   421→
   422→  const timestamps = result.timestamp || [];
   423→  const quotes = result.indicators?.quote?.[0] || {};
   424→
   425→  const points: HistoricalDataPoint[] = [];
   426→  for (let i = 0; i < timestamps.length; i++) {
   427→    const close = quotes.close?.[i];
   428→    const open = quotes.open?.[i];
   429→    const high = quotes.high?.[i];
   430→    const low = quotes.low?.[i];
   431→    const vol = quotes.volume?.[i];
   432→    if (close != null && close > 0) {
   433→      points.push({
   434→        date: new Date(timestamps[i] * 1000).toISOString().split("T")[0],
   435→        open: Math.round((open || close) * 100) / 100,
   436→        high: Math.round((high || close) * 100) / 100,
   437→        low: Math.round((low || close) * 100) / 100,
   438→        close: Math.round(close * 100) / 100,
   439→        volume: vol || 0,
   440→      });
   441→    }
   442→  }
   443→
   444→  const finalData = points.slice(-days);
   445→
   446→  // Update moving averages on cached quote
   447→  const closes = finalData.map(d => d.close);
   448→  const cachedQuote = quoteCache.get(nseSymbol);
   449→  if (cachedQuote) {
   450→    const dma50 = calculateSMA(closes, 50);
   451→    const dma200 = calculateSMA(closes, 200);
   452→    cachedQuote.data.fiftyDMA = dma50 ? Math.round(dma50 * 100) / 100 : null;
   453→    cachedQuote.data.twoHundredDMA = dma200 ? Math.round(dma200 * 100) / 100 : null;
   454→    cachedQuote.data.percentAbove50DMA = dma50 ? ((cachedQuote.data.price - dma50) / dma50) * 100 : null;
   455→    cachedQuote.data.percentAbove200DMA = dma200 ? ((cachedQuote.data.price - dma200) / dma200) * 100 : null;
   456→  }
   457→
   458→  histCache.set(cacheKey, { data: finalData, timestamp: Date.now() });
   459→  return finalData;
   460→}

   462→// ==================== SECTOR PEERS ====================
   463→export async function getSectorPeers(
   464→  nseSymbol: string,
   465→  sector: string
   466→): Promise<PeerData[]> {
   467→  let peerSymbols: string[] = [];
   468→  for (const [sec, symbols] of Object.entries(SECTOR_PEERS)) {
   469→    if (
   470→      sector.toLowerCase().includes(sec.toLowerCase()) ||
   471→      sec.toLowerCase().includes(sector.toLowerCase())
   472→    ) {
   473→      peerSymbols = symbols.filter((s) => s !== nseSymbol).slice(0, 10);
   474→      break;
   475→    }
   476→  }
   477→
   478→  if (peerSymbols.length < 2) return [];
   479→
   480→  const allPeers: PeerData[] = [];
   481→  // Fetch in batches of 3 to avoid rate limits
   482→  for (let i = 0; i < peerSymbols.length; i += 3) {
   483→    const batch = peerSymbols.slice(i, i + 3);
   484→    const results = await Promise.allSettled(batch.map((s) => getLiveQuote(s)));
   485→    for (let j = 0; j < results.length; j++) {
   486→      if (results[j].status === "fulfilled") {
   487→        const q = results[j].value;
   488→        allPeers.push({
   489→          symbol: q.symbol,
   490→          name: q.name,
   491→          price: q.price,
   492→          changePct: q.changePct,
   493→          marketCap: q.marketCap,
   494→          pe: q.pe,
   495→          pb: q.pb,
   496→          divYield: q.dividendYield,
   497→          roe: q.roe,
   498→          revenueGrowth: q.revenueGrowth,
   499→        });
   500→      }
   501→    }
   502→  }
   503→
   504→  return allPeers;
   505→}
