   102→// ==================== FORMATTERS ====================
   103→function formatINR(v: number): string {
   104→  if (v >= 1e12) return '₹' + (v / 1e12).toFixed(2) + ' T';
   105→  if (v >= 1e7) return '₹' + (v / 1e7).toFixed(2) + ' Cr';
   106→  if (v >= 1e5) return '₹' + (v / 1e5).toFixed(2) + ' L';
   107→  return '₹' + v.toLocaleString('en-IN', { minimumFractionDigits: 2, maximumFractionDigits: 2 });
   108→}
   109→function formatNum(v: number): string {
   110→  if (v >= 1e12) return (v / 1e12).toFixed(2) + 'T';
   111→  if (v >= 1e7) return (v / 1e7).toFixed(2) + ' Cr';
   112→  if (v >= 1e5) return (v / 1e5).toFixed(2) + ' L';
   113→  if (v >= 1e3) return (v / 1e3).toFixed(1) + 'K';
   114→  return v.toFixed(2);
   115→}
   116→function formatVol(v: number): string { return formatNum(v); }
   117→function formatDate(d: string): string {
   118→  return new Date(d).toLocaleDateString('en-IN', { day: '2-digit', month: 'short', year: '2-digit' });
   119→}
   120→function formatExpiry(d: string): string {
   121→  return new Date(d).toLocaleDateString('en-IN', { day: '2-digit', month: 'short', year: 'numeric' });
   122→}
   123→function getSignalColor(s: SignalType): string {
   124→  return { STRONG_BUY: 'text-emerald-400', BUY: 'text-green-400', HOLD: 'text-amber-400', SELL: 'text-orange-400', STRONG_SELL: 'text-red-400' }[s];
   125→}
   126→function getSignalBg(s: SignalType): string {
   127→  return { STRONG_BUY: 'bg-emerald-500/20 border-emerald-500/40 text-emerald-400', BUY: 'bg-green-500/20 border-green-500/40 text-green-400', HOLD: 'bg-amber-500/20 border-amber-500/40 text-amber-400', SELL: 'bg-orange-500/20 border-orange-500/40 text-orange-400', STRONG_SELL: 'bg-red-500/20 border-red-500/40 text-red-400' }[s];
   128→}
   129→function getSignalIcon(s: SignalType): string {
   130→  return { STRONG_BUY: '⬆⬆', BUY: '⬆', HOLD: '⏸', SELL: '⬇', STRONG_SELL: '⬇⬇' }[s];
   131→}
   132→function getTypeBadgeColor(t: string): string {
   133→  return { equity: 'bg-cyan-500/15 text-cyan-400 border-cyan-500/30', index: 'bg-purple-500/15 text-purple-400 border-purple-500/30', CE: 'bg-emerald-500/15 text-emerald-400 border-emerald-500/30', PE: 'bg-red-500/15 text-red-400 border-red-500/30' }[t] || 'bg-slate-500/15 text-slate-400 border-slate-500/30';
   134→}
