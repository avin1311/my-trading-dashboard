    37→interface BacktestResult { totalReturn: number; totalReturnPct: number; winRate: number; totalTrades: number; winningTrades: number; losingTrades: number; avgWinPct: number; avgLossPct: number; maxDrawdownPct: number; profitFactor: number; trades: TradeRecord[]; }
    38→interface StrategyParams { supertrendPeriod: number; supertrendMultiplier: number; rsiPeriod: number; rsiOverbought: number; rsiOversold: number; macdFast: number; macdSlow: number; macdSignal: number; }

    50→// ==================== FORMATTERS ====================
    51→function formatINR(v: number): string {
    52→  if (v >= 1e12) return '₹' + (v / 1e12).toFixed(2) + ' T';
    53→  if (v >= 1e7) return '₹' + (v / 1e7).toFixed(2) + ' Cr';
    54→  if (v >= 1e5) return '₹' + (v / 1e5).toFixed(2) + ' L';
    55→  return '₹' + v.toLocaleString('en-IN', { minimumFractionDigits: 2, maximumFractionDigits: 2 });
    56→}
    57→function formatNum(v: number): string {
    58→  if (v >= 1e12) return (v / 1e12).toFixed(2) + 'T';
    59→  if (v >= 1e7) return (v / 1e7).toFixed(2) + ' Cr';
    60→  if (v >= 1e5) return (v / 1e5).toFixed(2) + ' L';
    61→  if (v >= 1e3) return (v / 1e3).toFixed(1) + 'K';
    62→  return v.toFixed(2);
    63→}
    64→function formatVol(v: number): string { return formatNum(v); }
    65→function formatDate(d: string): string { return new Date(d).toLocaleDateString('en-IN', { day: '2-digit', month: 'short', year: '2-digit' }); }
    66→function formatExpiry(d: string): string { return new Date(d).toLocaleDateString('en-IN', { day: '2-digit', month: 'short', year: 'numeric' }); }
    67→function getSignalColor(s: SignalType): string { return { STRONG_BUY: 'text-emerald-400', BUY: 'text-green-400', HOLD: 'text-amber-400', SELL: 'text-orange-400', STRONG_SELL: 'text-red-400' }[s]; }
    68→function getSignalBg(s: SignalType): string { return { STRONG_BUY: 'bg-emerald-500/20 border-emerald-500/40 text-emerald-400', BUY: 'bg-green-500/20 border-green-500/40 text-green-400', HOLD: 'bg-amber-500/20 border-amber-500/40 text-amber-400', SELL: 'bg-orange-500/20 border-orange-500/40 text-orange-400', STRONG_SELL: 'bg-red-500/20 border-red-500/40 text-red-400' }[s]; }
    69→function getSignalIcon(s: SignalType): string { return { STRONG_BUY: '⬆⬆', BUY: '⬆', HOLD: '⏸', SELL: '⬇', STRONG_SELL: '⬇⬇' }[s]; }
    70→function getTypeBadgeColor(t: string): string { return { equity: 'bg-cyan-500/15 text-cyan-400 border-cyan-500/30', index: 'bg-purple-500/15 text-purple-400 border-purple-500/30', CE: 'bg-emerald-500/15 text-emerald-400 border-emerald-500/30', PE: 'bg-red-500/15 text-red-400 border-red-500/30' }[t] || 'bg-slate-500/15 text-slate-400 border-slate-500/30'; }
    71→
    72→// ==================== MARKET BAR ITEM ====================
    73→function MarketBarItem({ label, quote, accent }: { label: string; quote: LiveQuote | null; accent?: string }) {
    74→  if (!quote) return <div className="flex flex-col items-center px-3 py-1.5"><span className="text-[10px] text-slate-500 font-medium">{label}</span><Skeleton className="h-3.5 w-16 bg-slate-800 mt-0.5" /></div>;
    75→  const isUp = quote.changePct >= 0;
    76→  return (
    77→    <div className="flex flex-col items-center px-3 py-1.5">
    78→      <span className="text-[10px] text-slate-500 font-medium">{label}</span>
    79→      <span className="text-xs font-bold text-slate-200 font-mono">{quote.price.toLocaleString('en-IN', { maximumFractionDigits: 2 })}</span>
    80→      <span className={`text-[10px] font-semibold font-mono ${isUp ? 'text-emerald-400' : 'text-red-400'}`}>{isUp ? '+' : ''}{quote.changePct.toFixed(2)}%</span>
    81→    </div>
    82→  );
    83→}
