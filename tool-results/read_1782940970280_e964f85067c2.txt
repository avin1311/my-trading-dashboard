    47→// ==================== FORMATTERS ====================
    48→function fINR(v: number): string {
    49→  if (v >= 1e12) return '\u20B9' + (v / 1e12).toFixed(2) + ' T';
    50→  if (v >= 1e7) return '\u20B9' + (v / 1e7).toFixed(2) + ' Cr';
    51→  if (v >= 1e5) return '\u20B9' + (v / 1e5).toFixed(2) + ' L';
    52→  return '\u20B9' + v.toLocaleString('en-IN', { minimumFractionDigits: 2, maximumFractionDigits: 2 });
    53→}
    54→function fNum(v: number): string { if (v >= 1e12) return (v / 1e12).toFixed(2) + 'T'; if (v >= 1e7) return (v / 1e7).toFixed(2) + ' Cr'; if (v >= 1e5) return (v / 1e5).toFixed(2) + ' L'; if (v >= 1e3) return (v / 1e3).toFixed(1) + 'K'; return v.toFixed(2); }
    55→function fDate(d: string): string { return new Date(d).toLocaleDateString('en-IN', { day: '2-digit', month: 'short', year: '2-digit' }); }

    69→function pctVal(v: number | null) {
    70→  if (v === null) return <span className="text-slate-600">--</span>;
    71→  const c = v >= 0 ? 'text-emerald-400' : 'text-red-400';
    72→  const icon = v > 0.5 ? <ArrowUpRight className="w-3 h-3" /> : v < -0.5 ? <ArrowDownRight className="w-3 h-3" /> : <Minus className="w-3 h-3" />;
    73→  return <span className={cn(c, 'flex items-center gap-0.5 font-mono')}>{icon}{Math.abs(v).toFixed(2)}%</span>;
    74→}

   101→function PBar({ value, min, max, label, color }: { value: number; min: number; max: number; label: string; color: string }) {
   102→  const pct = max > min ? Math.max(0, Math.min(100, ((value - min) / (max - min)) * 100)) : 50;
   103→  return (
   104→    <div className="flex items-center gap-2 text-xs">
   105→      <span className="w-16 text-slate-500 text-right shrink-0">{label}</span>
   106→      <div className="flex-1 h-1.5 bg-slate-800 rounded-full overflow-hidden">
   107→        <div className={cn('h-full rounded-full', color)} style={{ width: pct + '%' }} />
   108→      </div>
   109→      <span className="w-12 text-slate-300 font-mono text-right shrink-0">{value.toFixed(1)}</span>
   110→    </div>
   111→  );
   112→}
