   170→   170→
   171→   171→function formatINR(value: number): string {
   172→   172→  return '₹' + value.toLocaleString('en-IN', {
   173→   173→    minimumFractionDigits: 2,
   174→   174→    maximumFractionDigits: 2,
   175→   175→  });
   176→   176→}
   177→   177→
   178→   178→function formatVolume(value: number): string {
   179→   179→  if (value >= 10000000) {
   180→   180→    return (value / 10000000).toFixed(2) + ' Cr';
   181→   181→  }
   182→   182→  if (value >= 100000) {
   183→   183→    return (value / 100000).toFixed(2) + ' L';
   184→   184→  }
   185→   185→  if (value >= 1000) {
   186→   186→    return (value / 1000).toFixed(1) + 'K';
   187→   187→  }
   188→   188→  return value.toString();
   189→   189→}
   190→   190→
   191→   191→function formatDate(dateStr: string): string {
   192→   192→  const d = new Date(dateStr);
   193→   193→  return d.toLocaleDateString('en-IN', {
   194→   194→    day: '2-digit',
   195→   195→    month: 'short',
   196→   196→    year: '2-digit',
   197→   197→  });
   198→   198→}

   200→   200→function getSignalColor(signal: SignalType): string {
   201→   201→  switch (signal) {
   202→   202→    case 'STRONG_BUY': return 'text-emerald-400';
   203→   203→    case 'BUY': return 'text-green-400';
   204→   204→    case 'HOLD': return 'text-amber-400';
   205→   205→    case 'SELL': return 'text-orange-400';
   206→   206→    case 'STRONG_SELL': return 'text-red-400';
   207→   207→  }
   208→   208→}
   209→   209→
   210→   210→function getSignalBg(signal: SignalType): string {
   211→   211→  switch (signal) {
   212→   212→    case 'STRONG_BUY': return 'bg-emerald-500/20 border-emerald-500/40 text-emerald-400';
   213→   213→    case 'BUY': return 'bg-green-500/20 border-green-500/40 text-green-400';
   214→   214→    case 'HOLD': return 'bg-amber-500/20 border-amber-500/40 text-amber-400';
   215→   215→    case 'SELL': return 'bg-orange-500/20 border-orange-500/40 text-orange-400';
   216→   216→    case 'STRONG_SELL': return 'bg-red-500/20 border-red-500/40 text-red-400';
