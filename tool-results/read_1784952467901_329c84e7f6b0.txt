   594→// ==================== SCREENER VIEW (Screener.in Style) ====================
   595→function ScreenerView({ d }: { d: ReturnType<typeof useDashboardData> }) {
   596→  const [sortBy, setSortBy] = useState<string>('changePct');
   597→  const [sortDir, setSortDir] = useState<'asc' | 'desc'>('desc');
   598→  const sorted = useMemo(() => {
   599→    const arr = [...d.filteredScreener];
   600→    arr.sort((a, b) => {
   601→      const va = (a as any)[sortBy]; const vb = (b as any)[sortBy];
   602→      if (typeof va === 'number' && typeof vb === 'number') return sortDir === 'desc' ? vb - va : va - vb;
   603→      return 0;
   604→    });
   605→    return arr;
   606→  }, [d.filteredScreener, sortBy, sortDir]);
   607→  const handleSort = (col: string) => { if (sortBy === col) setSortDir(p => p === 'asc' ? 'desc' : 'asc'); else { setSortBy(col); setSortDir('desc'); } };
   608→  return (
   609→    <P title="Full Stock Screener" icon={Search} source="Screener.in" className="h-full">
   610→      <div className="flex flex-wrap items-center gap-2 mb-3">
