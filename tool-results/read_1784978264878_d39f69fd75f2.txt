   686→// ==================== SCREENER VIEW (Screener.in Style) ====================
   687→function ScreenerView({ d }: { d: ReturnType<typeof useDashboardData> }) {
   688→  const [sortBy, setSortBy] = useState<string>('changePct');
   689→  const [sortDir, setSortDir] = useState<'asc' | 'desc'>('desc');
   690→  const sorted = useMemo(() => {
   691→    const arr = [...d.filteredScreener];
   692→    arr.sort((a, b) => {
   693→      const va = (a as any)[sortBy]; const vb = (b as any)[sortBy];
   694→      if (typeof va === 'number' && typeof vb === 'number') return sortDir === 'desc' ? vb - va : va - vb;
   695→      return 0;
   696→    });
   697→    return arr;
   698→  }, [d.filteredScreener, sortBy, sortDir]);
   699→  const handleSort = (col: string) => { if (sortBy === col) setSortDir(p => p === 'asc' ? 'desc' : 'asc'); else { setSortBy(col); setSortDir('desc'); } };
   700→  return (
   701→    <P title="Full Stock Screener" icon={Search} source="Screener.in" className="h-full">
   702→      <div className="flex flex-wrap items-center gap-2 mb-3">
   703→        <Button size="sm" className="h-8 text-xs bg-emerald-600 hover:bg-emerald-500" onClick={d.fetchScreener} disabled={d.screenerLoading}>
   704→          {d.screenerLoading ? <RefreshCw className="w-3.5 h-3.5 mr-1.5 animate-spin" /> : <Zap className="w-3.5 h-3.5 mr-1.5" />}

   708→          <SelectTrigger className="h-8 w-[150px] text-xs"><SelectValue /></SelectTrigger>
   709→          <SelectContent>
   710→            <SelectItem value="ALL">All Signals</SelectItem>

   717→            <Badge key={sig} variant="outline" className={cn('text-[9px] px-2 py-0.5 cursor-pointer hover:opacity-80', SIG_BG[sig as keyof typeof SIG_BG] || SIG_BG.HOLD)} onClick={() => { d.setScreenerFilter(sig); d.setScreenerData([]); d.fetchScreener(); }}>{cnt} {sig.replace('_', ' ')}</Badge>
   718→          ))}

   722→        <Table>
   723→          <TableHeader><TableRow className="border-slate-800 hover:bg-transparent">
   724→            {([['symbol', 'Stock'], ['price', 'Price'], ['changePct', 'Change'], ['rsi', 'RSI'], ['signal', 'Signal'], ['volume', 'Volume'], ['marketCap', 'Mkt Cap'], ['pe', 'P/E']] as [string, string][]).map(([key, label]) => (
   725→              <TableHead key={key} className={cn('text-[10px] text-slate-400 h-8 cursor-pointer hover:text-white', key === 'price' || key === 'changePct' || key === 'rsi' || key === 'volume' || key === 'marketCap' || key === 'pe' ? 'text-right' : '')} onClick={() => handleSort(key)}>
   726→                {label} {sortBy === key && <span className="text-emerald-400 ml-0.5">{sortDir === 'desc' ? '↓' : '↑'}</span>}

  1075→// ==================== WATCHLIST VIEW ====================
  1076→function WatchlistView({ d, watchlist }: { d: ReturnType<typeof useDashboardData>; watchlist: ReturnType<typeof useWatchlist> }) {
  1077→  const [watchlistQuotes, setWatchlistQuotes] = useState<Record<string, { price: number; changePct: number; name: string; loading: boolean }>>({});
  1078→  useEffect(() => {
  1079→    if (watchlist.watchlist.length === 0) return;
  1080→    const fetchQuotes = async () => {
  1081→      const results: typeof watchlistQuotes = {};
  1082→      for (const item of watchlist.watchlist) {
  1083→        results[item.symbol] = { price: 0, changePct: 0, name: item.symbol, loading: true };
  1084→      }
  1085→      setWatchlistQuotes({ ...results });
  1086→      for (const item of watchlist.watchlist) {
  1087→        try {
  1088→          const res = await fetch('/api/stock-detail?symbol=' + item.symbol);
  1089→          const data = await res.json();
  1090→          if (data.quote) {
  1091→            results[item.symbol] = { price: data.quote.price, changePct: data.quote.changePct, name: data.quote.longName || data.quote.name, loading: false };
  1092→          }
  1093→        } catch { results[item.symbol] = { ...results[item.symbol], loading: false }; }
  1094→        setWatchlistQuotes({ ...results });
  1095→      }
  1096→    };
  1097→    fetchQuotes();
  1098→    const interval = setInterval(fetchQuotes, 30000);
  1099→    return () => clearInterval(interval);
  1100→  }, [watchlist.watchlist]);
  1101→  return (
  1102→    <P title="My Watchlist" icon={Star} badge={<Badge variant="outline" className="text-[10px] px-2 py-0.5 bg-amber-500/10 border-amber-500/30 text-amber-400">{watchlist.watchlist.length} stocks</Badge>} source="Custom">
  1103→      {watchlist.watchlist.length === 0 ? (
  1104→        <div className="text-center py-16">
  1105→          <Star className="w-12 h-12 mx-auto mb-3 text-slate-700" />
  1106→          <p className="text-sm text-slate-400">Your watchlist is empty</p>
  1107→          <p className="text-xs text-slate-600 mt-1">Click the star icon on any stock to add it here</p>
  1108→        </div>
  1109→      ) : (
  1110→        <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 xl:grid-cols-4 gap-3">
  1111→          {watchlist.watchlist.map(item => {
  1112→            const info = watchlistQuotes[item.symbol];
  1113→            return (
  1114→              <div key={item.symbol} className="p-3 rounded-xl bg-slate-800/20 border border-slate-800/40 hover:bg-slate-800/40 cursor-pointer transition-colors" onClick={() => d.handleSelect(item.symbol, item.type)}>
  1115→                <div className="flex items-center justify-between mb-2">
  1116→                  <div>
  1117→                    <div className="text-sm font-bold text-slate-200">{item.symbol}</div>
  1118→                    <div className="text-[10px] text-slate-500 truncate max-w-[140px]">{info?.name || item.symbol}</div>
  1119→                  </div>
  1120→                  <button onClick={(e) => { e.stopPropagation(); watchlist.removeFromWatchlist(item.symbol); }} className="p-1 rounded hover:bg-red-500/20 text-slate-600 hover:text-red-400">
  1121→                    <X className="w-3.5 h-3.5" />
  1122→                  </button>
  1123→                </div>
  1124→                {info?.loading ? <Skeleton className="h-6 w-24 bg-slate-700" /> : info?.price ? (
  1125→                  <div className="flex items-end justify-between">
  1126→                    <span className="text-lg font-bold font-mono text-slate-100">{info.price.toLocaleString('en-IN', { maximumFractionDigits: 2 })}</span>
  1127→                    <span className={cn('text-sm font-semibold font-mono', info.changePct >= 0 ? 'text-emerald-400' : 'text-red-400')}>
  1128→                      {info.changePct >= 0 ? '+' : ''}{info.changePct.toFixed(2)}%
  1129→                    </span>
  1130→                  </div>
  1131→                ) : <Skeleton className="h-6 w-24 bg-slate-700" />}
  1132→              </div>
  1133→            );
  1134→          })}
  1135→        </div>
  1136→      )}
  1137→    </P>
  1138→  );
  1139→}
