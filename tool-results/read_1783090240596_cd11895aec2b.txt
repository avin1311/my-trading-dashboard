   923→// ==================== WATCHLIST VIEW ====================
   924→function WatchlistView({ d, watchlist }: { d: ReturnType<typeof useDashboardData>; watchlist: ReturnType<typeof useWatchlist> }) {
   925→  const [watchlistQuotes, setWatchlistQuotes] = useState<Record<string, { price: number; changePct: number; name: string; loading: boolean }>>({});
   926→  useEffect(() => {
   927→    if (watchlist.watchlist.length === 0) return;
   928→    const fetchQuotes = async () => {
   929→      const results: typeof watchlistQuotes = {};
   930→      for (const item of watchlist.watchlist) {
   931→        results[item.symbol] = { price: 0, changePct: 0, name: item.symbol, loading: true };
   932→      }
   933→      setWatchlistQuotes({ ...results });
   934→      for (const item of watchlist.watchlist) {
   935→        try {
   936→          const res = await fetch('/api/stock-detail?symbol=' + item.symbol);
   937→          const data = await res.json();
   938→          if (data.quote) {
   939→            results[item.symbol] = { price: data.quote.price, changePct: data.quote.changePct, name: data.quote.longName || data.quote.name, loading: false };
   940→          }
   941→        } catch { results[item.symbol] = { ...results[item.symbol], loading: false }; }
   942→        setWatchlistQuotes({ ...results });
   943→      }
   944→    };
   945→    fetchQuotes();
   946→    const interval = setInterval(fetchQuotes, 30000);
   947→    return () => clearInterval(interval);
   948→  }, [watchlist.watchlist]);
   949→  return (
   950→    <P title="My Watchlist" icon={Star} badge={<Badge variant="outline" className="text-[10px] px-2 py-0.5 bg-amber-500/10 border-amber-500/30 text-amber-400">{watchlist.watchlist.length} stocks</Badge>} source="Custom">
   951→      {watchlist.watchlist.length === 0 ? (
   952→        <div className="text-center py-16">
   953→          <Star className="w-12 h-12 mx-auto mb-3 text-slate-700" />
   954→          <p className="text-sm text-slate-400">Your watchlist is empty</p>
   955→          <p className="text-xs text-slate-600 mt-1">Click the star icon on any stock to add it here</p>
   956→        </div>
   957→      ) : (
   958→        <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 xl:grid-cols-4 gap-3">
   959→          {watchlist.watchlist.map(item => {
   960→            const info = watchlistQuotes[item.symbol];
   961→            return (
   962→              <div key={item.symbol} className="p-3 rounded-xl bg-slate-800/20 border border-slate-800/40 hover:bg-slate-800/40 cursor-pointer transition-colors" onClick={() => d.handleSelect(item.symbol, item.type)}>
   963→                <div className="flex items-center justify-between mb-2">
   964→                  <div>
   965→                    <div className="text-sm font-bold text-slate-200">{item.symbol}</div>
   966→                    <div className="text-[10px] text-slate-500 truncate max-w-[140px]">{info?.name || item.symbol}</div>
   967→                  </div>
   968→                  <button onClick={(e) => { e.stopPropagation(); watchlist.removeFromWatchlist(item.symbol); }} className="p-1 rounded hover:bg-red-500/20 text-slate-600 hover:text-red-400">
   969→                    <X className="w-3.5 h-3.5" />
   970→                  </button>
   971→                </div>
   972→                {info?.loading ? <Skeleton className="h-6 w-24 bg-slate-700" /> : info?.price ? (
   973→                  <div className="flex items-end justify-between">
   974→                    <span className="text-lg font-bold font-mono text-slate-100">{info.price.toLocaleString('en-IN', { maximumFractionDigits: 2 })}</span>
   975→                    <span className={cn('text-sm font-semibold font-mono', info.changePct >= 0 ? 'text-emerald-400' : 'text-red-400')}>
   976→                      {info.changePct >= 0 ? '+' : ''}{info.changePct.toFixed(2)}%
   977→                    </span>
   978→                  </div>
   979→                ) : <Skeleton className="h-6 w-24 bg-slate-700" />}
   980→              </div>
   981→            );
   982→          })}
   983→        </div>
   984→      )}
   985→    </P>
   986→  );
   987→}
