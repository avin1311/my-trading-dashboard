   552→// ==================== SCREENER VIEW (Screener.in Style) ====================
   553→function ScreenerView({ d }: { d: ReturnType<typeof useDashboardData> }) {
   554→  const [sortBy, setSortBy] = useState<string>('changePct');

   557→    const arr = [...d.filteredScreener];

   559→      const va = (a as any)[sortBy]; const vb = (b as any)[sortBy];

   562→    });

   941→function WatchlistView({ d, watchlist }: { d: ReturnType<typeof useDashboardData>; watchlist: ReturnType<typeof useWatchlist> }) {
   942→  const [watchlistQuotes, setWatchlistQuotes] = useState<Record<string, { price: number; changePct: number; name: string; loading: boolean }>>({});
   943→  useEffect(() => {
   944→    if (watchlist.watchlist.length === 0) return;
   945→    const fetchQuotes = async () => {
   946→      const results: typeof watchlistQuotes = {};
   947→      for (const item of watchlist.watchlist) {
   948→        results[item.symbol] = { price: 0, changePct: 0, name: item.symbol, loading: true };
   949→      }
   950→      setWatchlistQuotes({ ...results });
   951→      for (const item of watchlist.watchlist) {
   952→        try {
   953→          const res = await fetch('/api/stock-detail?symbol=' + item.symbol);
   954→          const data = await res.json();
   955→          if (data.quote) {
   956→            results[item.symbol] = { price: data.quote.price, changePct: data.quote.changePct, name: data.quote.longName || data.quote.name, loading: false };
   957→          }
   958→        } catch { results[item.symbol] = { ...results[item.symbol], loading: false }; }
   959→        setWatchlistQuotes({ ...results });
   960→      }
   961→    };
   962→    fetchQuotes();
   963→    const interval = setInterval(fetchQuotes, 30000);
   964→    return () => clearInterval(interval);
   965→  }, [watchlist.watchlist]);
   966→  return (
   967→    <P title="My Watchlist" icon={Star} badge={<Badge variant="outline" className="text-[10px] px-2 py-0.5 bg-amber-500/10 border-amber-500/30 text-amber-400">{watchlist.watchlist.length} stocks</Badge>} source="Custom">
   968→      {watchlist.watchlist.length === 0 ? (
   969→        <div className="text-center py-16">
   970→          <Star className="w-12 h-12 mx-auto mb-3 text-slate-700" />
   971→          <p className="text-sm text-slate-400">Your watchlist is empty</p>
   972→          <p className="text-xs text-slate-600 mt-1">Click the star icon on any stock to add it here</p>
   973→        </div>
   974→      ) : (
   975→        <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 xl:grid-cols-4 gap-3">
   976→          {watchlist.watchlist.map(item => {
   977→            const info = watchlistQuotes[item.symbol];
   978→            return (
   979→              <div key={item.symbol} className="p-3 rounded-xl bg-slate-800/20 border border-slate-800/40 hover:bg-slate-800/40 cursor-pointer transition-colors" onClick={() => d.handleSelect(item.symbol, item.type)}>
   980→                <div className="flex items-center justify-between mb-2">
   981→                  <div>
   982→                    <div className="text-sm font-bold text-slate-200">{item.symbol}</div>
   983→                    <div className="text-[10px] text-slate-500 truncate max-w-[140px]">{info?.name || item.symbol}</div>
   984→                  </div>
   985→                  <button onClick={(e) => { e.stopPropagation(); watchlist.removeFromWatchlist(item.symbol); }} className="p-1 rounded hover:bg-red-500/20 text-slate-600 hover:text-red-400">
   986→                    <X className="w-3.5 h-3.5" />
   987→                  </button>
   988→                </div>

   990→                  <div className="flex items-end justify-between">
   991→                    <span className="text-lg font-bold font-mono text-slate-100">{info.price.toLocaleString('en-IN', { maximumFractionDigits: 2 })}</span>
