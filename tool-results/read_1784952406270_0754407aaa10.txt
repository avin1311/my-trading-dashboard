    46→// ==================== FORMATTERS ====================
    47→function fINR(v: number): string {
    48→  if (v >= 1e12) return '\u20B9' + (v / 1e12).toFixed(2) + ' T';
    49→  if (v >= 1e7) return '\u20B9' + (v / 1e7).toFixed(2) + ' Cr';
    50→  if (v >= 1e5) return '\u20B9' + (v / 1e5).toFixed(2) + ' L';
    51→  return '\u20B9' + v.toLocaleString('en-IN', { minimumFractionDigits: 2, maximumFractionDigits: 2 });
    52→}
    53→function fNum(v: number): string { if (v >= 1e12) return (v / 1e12).toFixed(2) + 'T'; if (v >= 1e7) return (v / 1e7).toFixed(2) + ' Cr'; if (v >= 1e5) return (v / 1e5).toFixed(2) + ' L'; if (v >= 1e3) return (v / 1e3).toFixed(1) + 'K'; return v.toFixed(2); }
    54→function fDate(d: string): string { return new Date(d).toLocaleDateString('en-IN', { day: '2-digit', month: 'short', year: '2-digit' }); }

    68→function pctVal(v: number | null) {
    69→  if (v === null) return <span className="text-slate-600">--</span>;
    70→  const c = v >= 0 ? 'text-emerald-400' : 'text-red-400';
    71→  const icon = v > 0.5 ? <ArrowUpRight className="w-3 h-3" /> : v < -0.5 ? <ArrowDownRight className="w-3 h-3" /> : <Minus className="w-3 h-3" />;
    72→  return <span className={cn(c, 'flex items-center gap-0.5 font-mono')}>{icon}{Math.abs(v).toFixed(2)}%</span>;
    73→}

   155→function mergeChartData(stockData: OHLCV[], signals: StrategySignal[]): ChartDataPoint[] {
   156→  const sigMap = new Map(signals.map(s => [s.date, s]));
   157→  return stockData.map(d => {
   158→    const sig = sigMap.get(d.date);
   159→    return { ...d, supertrend: sig?.supertrend ?? null, supertrendDir: sig?.supertrendDir ?? null, rsi: sig?.rsi ?? null, macd: sig?.macd ?? null, macdSignal: sig?.macdSignal ?? null, macdHistogram: sig?.macdHistogram ?? null, signal: sig?.signal ?? null };
   160→  });
   161→}
