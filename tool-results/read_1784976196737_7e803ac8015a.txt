   685→// ==================== SCREENER VIEW (Screener.in Style) ====================
   686→function ScreenerView({ d }: { d: ReturnType<typeof useDashboardData> }) {
   687→  const [sortBy, setSortBy] = useState<string>('changePct');
   688→  const [sortDir, setSortDir] = useState<'asc' | 'desc'>('desc');
   689→  const sorted = useMemo(() => {
   690→    const arr = [...d.filteredScreener];
   691→    arr.sort((a, b) => {
   692→      const va = (a as any)[sortBy]; const vb = (b as any)[sortBy];
   693→      if (typeof va === 'number' && typeof vb === 'number') return sortDir === 'desc' ? vb - va : va - vb;
   694→      return 0;
   695→    });
   696→    return arr;
   697→  }, [d.filteredScreener, sortBy, sortDir]);
   698→  const handleSort = (col: string) => { if (sortBy === col) setSortDir(p => p === 'asc' ? 'desc' : 'asc'); else { setSortBy(col); setSortDir('desc'); } };
   699→  return (
   700→    <P title="Full Stock Screener" icon={Search} source="Screener.in" className="h-full">
   701→      <div className="flex flex-wrap items-center gap-2 mb-3">
   702→        <Button size="sm" className="h-8 text-xs bg-emerald-600 hover:bg-emerald-500" onClick={d.fetchScreener} disabled={d.screenerLoading}>
   703→          {d.screenerLoading ? <RefreshCw className="w-3.5 h-3.5 mr-1.5 animate-spin" /> : <Zap className="w-3.5 h-3.5 mr-1.5" />}
   704→          {d.screenerLoading ? 'Scanning All Stocks...' : 'Scan All Stocks'}

   707→          <SelectTrigger className="h-8 w-[150px] text-xs"><SelectValue /></SelectTrigger>

   709→            <SelectItem value="ALL">All Signals</SelectItem>
   710→            {Object.entries(d.screenerCounts).map(([sig, cnt]) => <SelectItem key={sig} value={sig}>{sig.replace('_', ' ')} ({cnt})</SelectItem>)}

   716→            <Badge key={sig} variant="outline" className={cn('text-[9px] px-2 py-0.5 cursor-pointer hover:opacity-80', SIG_BG[sig as keyof typeof SIG_BG] || SIG_BG.HOLD)} onClick={() => { d.setScreenerFilter(sig); d.setScreenerData([]); d.fetchScreener(); }}>{cnt} {sig.replace('_', ' ')}</Badge>
   717→          ))}

   722→          <TableHeader><TableRow className="border-slate-800 hover:bg-transparent">
   723→            {([['symbol', 'Stock'], ['price', 'Price'], ['changePct', 'Change'], ['rsi', 'RSI'], ['signal', 'Signal'], ['volume', 'Volume'], ['marketCap', 'Mkt Cap'], ['pe', 'P/E']] as [string, string][]).map(([key, label]) => (
   724→              <TableHead key={key} className={cn('text-[10px] text-slate-400 h-8 cursor-pointer hover:text-white', key === 'price' || key === 'changePct' || key === 'rsi' || key === 'volume' || key === 'marketCap' || key === 'pe' ? 'text-right' : '')} onClick={() => handleSort(key)}>
   725→                {label} {sortBy === key && <span className="text-emerald-400 ml-0.5">{sortDir === 'desc' ? '↓' : '↑'}</span>}

   728→          </TableRow></TableHeader>

  1074→// ==================== WATCHLIST VIEW ====================
  1075→function WatchlistView({ d, watchlist }: { d: ReturnType<typeof useDashboardData>; watchlist: ReturnType<typeof useWatchlist> }) {
  1076→  const [watchlistQuotes, setWatchlistQuotes] = useState<Record<string, { price: number; changePct: number; name: string; loading: boolean }>>({});
  1077→  useEffect(() => {
  1078→    if (watchlist.watchlist.length === 0) return;
  1079→    const fetchQuotes = async () => {
  1080→      const results: typeof watchlistQuotes = {};
  1081→      for (const item of watchlist.watchlist) {
  1082→        results[item.symbol] = { price: 0, changePct: 0, name: item.symbol, loading: true };
  1083→      }
  1084→      setWatchlistQuotes({ ...results });
  1085→      for (const item of watchlist.watchlist) {
  1086→        try {
  1087→          const res = await fetch('/api/stock-detail?symbol=' + item.symbol);
  1088→          const data = await res.json();
  1089→          if (data.quote) {
  1090→            results[item.symbol] = { price: data.quote.price, changePct: data.quote.changePct, name: data.quote.longName || data.quote.name, loading: false };
  1091→          }
  1092→        } catch { results[item.symbol] = { ...results[item.symbol], loading: false }; }
  1093→        setWatchlistQuotes({ ...results });
  1094→      }
  1095→    };
  1096→    fetchQuotes();
  1097→    const interval = setInterval(fetchQuotes, 30000);
  1098→    return () => clearInterval(interval);
  1099→  }, [watchlist.watchlist]);
  1100→  return (
  1101→    <P title="My Watchlist" icon={Star} badge={<Badge variant="outline" className="text-[10px] px-2 py-0.5 bg-amber-500/10 border-amber-500/30 text-amber-400">{watchlist.watchlist.length} stocks</Badge>} source="Custom">
  1102→      {watchlist.watchlist.length === 0 ? (
  1103→        <div className="text-center py-16">
  1104→          <Star className="w-12 h-12 mx-auto mb-3 text-slate-700" />
  1105→          <p className="text-sm text-slate-400">Your watchlist is empty</p>
  1106→          <p className="text-xs text-slate-600 mt-1">Click the star icon on any stock to add it here</p>
  1107→        </div>
  1108→      ) : (
  1109→        <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 xl:grid-cols-4 gap-3">
  1110→          {watchlist.watchlist.map(item => {
  1111→            const info = watchlistQuotes[item.symbol];
  1112→            return (
  1113→              <div key={item.symbol} className="p-3 rounded-xl bg-slate-800/20 border border-slate-800/40 hover:bg-slate-800/40 cursor-pointer transition-colors" onClick={() => d.handleSelect(item.symbol, item.type)}>
  1114→                <div className="flex items-center justify-between mb-2">
  1115→                  <div>
  1116→                    <div className="text-sm font-bold text-slate-200">{item.symbol}</div>
  1117→                    <div className="text-[10px] text-slate-500 truncate max-w-[140px]">{info?.name || item.symbol}</div>
  1118→                  </div>
  1119→                  <button onClick={(e) => { e.stopPropagation(); watchlist.removeFromWatchlist(item.symbol); }} className="p-1 rounded hover:bg-red-500/20 text-slate-600 hover:text-red-400">
  1120→                    <X className="w-3.5 h-3.5" />
  1121→                  </button>
  1122→                </div>
  1123→                {info?.loading ? <Skeleton className="h-6 w-24 bg-slate-700" /> : info?.price ? (
  1124→                  <div className="flex items-end justify-between">
  1125→                    <span className="text-lg font-bold font-mono text-slate-100">{info.price.toLocaleString('en-IN', { maximumFractionDigits: 2 })}</span>
  1126→                    <span className={cn('text-sm font-semibold font-mono', info.changePct >= 0 ? 'text-emerald-400' : 'text-red-400')}>
  1127→                      {info.changePct >= 0 ? '+' : ''}{info.changePct.toFixed(2)}%
  1128→                    </span>
  1129→                  </div>
  1130→                ) : <Skeleton className="h-6 w-24 bg-slate-700" />}
  1131→              </div>
  1132→            );
  1133→          })}
  1134→        </div>
  1135→      )}
  1136→    </P>
  1137→  );
  1138→}
